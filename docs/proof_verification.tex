\section{Setup and standing conventions}

Population of size $N$; deaths $D$ in the window $[t_0,t_1]$. Classes: $W$ (women of all ages $+$ males under 18) with population share $w$, and $\mathrm{AM}$ (\emph{all} males 18$+$) with population share $a$, so
\[
w + a = 1 \quad\text{exactly.}
\]
The exact partition matters: the identification formulas substitute $a=1-w$, so any convention in which $\mathrm{AM}$ excludes part of the male population (e.g.\ men over 60) silently breaks them.

Combatant stock $M$, $f = M/N$, all combatants in $\mathrm{AM}$ (slack for exceptions handled in Claim~\ref{cl:remark}). Estimand $q = D_M/D$, the combatant share of the dead; $\omega$ = share of the \emph{dead} in class $W$.

\paragraph{The hazard model.} All identification results use one structural model: conditional on being a civilian, a member of class $W$ dies with per-capita hazard $\lambda$ and a member of civilian-$\mathrm{AM}$ with hazard $\mu\lambda$, $\mu\ge 1$. Combatant deaths are governed separately (their total is $qD$ by definition). No assumption is made about $\lambda$ itself --- it cancels.

\section{Claim 1: benchmark identification (paper eq.~2, Assumption 3.1)}

\begin{claim}[Point identification under exchangeable collateral]\label{cl:ident}
If all civilians face equal per-capita hazard ($\mu=1$), then
\[
q = 1-\frac{\omega(1-f)}{w}.
\]
\end{claim}

\begin{proof}
Civilian population masses: class $W$ has $wN$ civilians (no combatants in $W$); civilian $\mathrm{AM}$ has $(a-f)N$. With equal hazard $\lambda$, expected civilian deaths in $W$ and civilian-$\mathrm{AM}$ are $\lambda wN$ and $\lambda(a-f)N$, so the fraction of \emph{civilian} deaths landing in $W$ is
\[
\frac{wN}{wN+(a-f)N} = \frac{w}{w+a-f} = \frac{w}{1-f},
\]
where the last step uses $w+a=1$. Combatant deaths (share $q$ of all deaths) land entirely in $\mathrm{AM}$. Hence the share of \emph{all} deaths in $W$ is
\[
\omega = (1-q)\cdot\frac{w}{1-f},
\]
and solving for $q$ gives the display. Note this is exact in expectation (a first-moment identity), not an approximation; finite-sample noise is handled by the delta method (Claim~\ref{cl:delta}).
\end{proof}

\begin{verification}
\PASS. Sympy: solving $\omega=(1-q)w\lambda/(w\lambda+(a-f)\mu\lambda)$ for $q$ and substituting $\mu=1$, $a=1-w$ reproduces the display exactly ($\lambda$ cancels symbolically). Lean: \texttt{claim1\_benchmark\_identification}. Degenerate cases: $\omega=w$, $f=0$ $\Rightarrow$ $q=0$; $\omega=0$ $\Rightarrow$ $q=1$. \emph{Subtlety for manual checking:} the last step of the proof needs $w+a=1$ exactly (see the convention note in Section~1); with $a<1-w$ the two displayed forms of the identity disagree.
\end{verification}

\section{Claim 2: sharp identified set (paper Theorem 3.3)}

\begin{claim}[Sharp identified set under $\mu$-bounded exposure]\label{cl:sharp}
Suppose $f\le a$ and the hazard model holds for some $\mu\in[\mu_{\mathrm{lo}},\mu_{\mathrm{hi}}]$. Given $(\omega,w,a,f)$,
\[
q(\mu) = 1-\omega\,\frac{w+\mu(a-f)}{w},
\]
is monotone decreasing in $\mu$, and the identified set for $q$ is exactly
$[\,q(\mu_{\mathrm{hi}}),\,q(\mu_{\mathrm{lo}})\,]\cap[0,1]$.
\end{claim}

\begin{proof}
\emph{Step 1 (the curve).} With hazards $(\lambda,\mu\lambda)$, expected civilian deaths are proportional to $\lambda wN + \mu\lambda(a-f)N$, so the civilian share landing in $W$ is $w/(w+\mu(a-f))$ and
\begin{equation}\label{eq:core}
\omega = (1-q)\,\frac{w}{w+\mu(a-f)}.
\end{equation}
Solving \eqref{eq:core} for $q$ gives $q(\mu)$.

\emph{Step 2 (monotonicity).} $\partial q/\partial\mu = -\omega(a-f)/w \le 0$, with equality iff $a=f$ (no civilian adult men) or $\omega=0$.

\emph{Step 3 (validity: every point of the demographic interval with $q(\mu^\ast)\in[0,1]$ is attainable).} Fix any $\mu^\ast\in[\mu_{\mathrm{lo}},\mu_{\mathrm{hi}}]$ such that $q(\mu^\ast)\in(0,1)$. With $(D,N)$ fixed, the observed $W$-class deaths pin $\lambda=\omega D/(wN)$. The theorem's feasibility conditions imply $\lambda\le1$ and $\mu^\ast\lambda\le1$, so the civilian class death counts do not exceed their populations. The resulting demographic allocation satisfies \eqref{eq:core} with $q=q(\mu^\ast)$. Boundary points are attained directly: $q(\mu^\ast)=1$ forces $\omega=0$ and hence $\lambda=0$; $q(\mu^\ast)=0$ is attained with no combatant deaths. Hence $q(\mu^\ast)$ is consistent with the demographic data. The separate stock restriction $D_M\le M$ is imposed in Claim~\ref{cl:manpower}; before that intersection, continuity and monotonicity give $[q(\mu_{\mathrm{hi}}),q(\mu_{\mathrm{lo}})]\cap[0,1]$.

\emph{Step 4 (sharpness: nothing outside is attainable).} Conversely, any admissible configuration has some $\mu\in[\mu_{\mathrm{lo}},\mu_{\mathrm{hi}}]$, and \eqref{eq:core} then forces $q=q(\mu)$, which lies in the interval. The intersection with $[0,1]$ imposes the logical range of a share.
\end{proof}

\begin{verification}
\PASS. Sympy confirms the inversion and the derivative; Lean: \texttt{claim2\_inversion}, \texttt{claim2\_monotone}. 2{,}000 random parameter draws confirm endpoint ordering. The forward model's three death-shares (combatant, civilian-$\mathrm{AM}$, $W$) sum to 1 symbolically. \emph{Caveat worth knowing when checking manually:} the theorem treats $\omega$ as an exact population quantity; the ``identified set'' is a set for the estimand given error-free inputs. Sampling error enters separately via Claim~\ref{cl:delta} and systematic error via the anchor sensitivity analysis. This is stated in the paper's Discussion (limitation 3).
\end{verification}

\section{Claim 3: combatants outside AM (paper Remark 3)}

\begin{claim}[$\eta$-slack]\label{cl:remark}
Suppose a fraction $\eta\in[0,\bar\eta]$, $\bar\eta=0.03$, of combatant \emph{deaths} is recorded in class $W$, while the combatant stock stays in $\mathrm{AM}$ (a death-classification slack, so $S:=w/(w+\mu(a-f))$ is unchanged). The observable identity becomes $\omega=(1-q)S+\eta q$, so
\[
q_{\mathrm{true}} = \frac{S-\omega}{S-\eta},
\qquad
q_{\mathrm{true}}-q_{\eta=0} \;=\; \frac{\eta(S-\omega)}{S(S-\eta)}
\;=\; \frac{\eta\,q_{\mathrm{true}}}{S}
\;=\; \frac{\eta\,q_{\eta=0}}{S-\eta},
\]
and the bias is $\ge 0$ whenever $S>0$ and $q_{\mathrm{true}}\ge 0$: ignoring the slack \emph{understates} $q$. Provided $S>\bar\eta$ (which holds at the Gaza values, where $S\ge 0.626$ for $\mu\le 2$), a loose uniform bound over $\mu\in[1,2]$ with $q_{\eta=0}\le 0.33$ is $\bar\eta\cdot 0.33/(0.626-\bar\eta)\approx 1.7$ percentage points; the exact per-$\mu$ bias is smaller and decreasing in $\mu$ ($1.30$ points at $\mu=1$, $1.10$ at $\mu=1.5$, $0.83$ at $\mu=2$), because $q_{\eta=0}$ falls with $\mu$ faster than $S$ does.
\end{claim}

\begin{proof}
Deaths in $W$ now come from two sources: civilian deaths landing in $W$, which is $(1-q)S$ of all deaths as in Claim~\ref{cl:sharp}, plus the fraction $\eta$ of combatant deaths, contributing $\eta q$. Setting $\omega=(1-q)S+\eta q=S-q(S-\eta)$ and solving for $q$ gives $q_{\mathrm{true}}=(S-\omega)/(S-\eta)$. Subtracting $q_{\eta=0}=(S-\omega)/S$,
\[
q_{\mathrm{true}}-q_{\eta=0}
=(S-\omega)\Bigl(\frac{1}{S-\eta}-\frac{1}{S}\Bigr)
=\frac{\eta(S-\omega)}{S(S-\eta)}.
\]
Substituting $S-\omega=q_{\mathrm{true}}(S-\eta)$ gives the second form $\eta q_{\mathrm{true}}/S$; substituting $S-\omega=q_{\eta=0}S$ gives the third form $\eta q_{\eta=0}/(S-\eta)$.

\emph{Nonnegativity.} From the second form, $q_{\mathrm{true}}-q_{\eta=0}=\eta q_{\mathrm{true}}/S\ge 0$ for $S>0$, $\eta\ge 0$, $q_{\mathrm{true}}\ge 0$. (This is the right route: arguing instead from ``$\omega\le S$'' requires $S\ge\eta$, since $\omega=S-q(S-\eta)$ gives $\omega\ge S$ when $\eta>S$ and $q\ge 0$.)

\emph{Bound.} For the uniform bound the third form is used with the worst admissible values: it is increasing in $\eta$ (for $S>\eta$) and in $q_{\eta=0}$, and decreasing in $S$, so on $\eta\le\bar\eta$, $q_{\eta=0}\le 0.33$, $S\ge 0.626$ (the $\mu=2$ value; $S$ decreases in $\mu$) it is at most $\bar\eta\cdot 0.33/(0.626-\bar\eta)<0.017$. This is loose because it pairs the smallest $S$ (attained at $\mu=2$) with the largest $q_{\eta=0}$ (attained at $\mu=1$): the exact per-$\mu$ bias $\bar\eta\,q_{\eta=0}(\mu)/(S(\mu)-\bar\eta)$ evaluates to $0.0130$, $0.0110$, $0.0083$ at $\mu=1,1.5,2$.

Note the distinction between the second and third forms: the bias equals $\eta q/(S-\eta)$ with $q$ the \emph{no-slack} estimate, and $\eta q/S$ with $q$ the \emph{true} share --- the two coincide only to first order in $\eta$.
\end{proof}

\begin{verification}
\PASS. Sympy: all three displayed forms are symbolically equal to $q_{\mathrm{true}}-q_{\eta=0}$; the superficially plausible chain $\eta q_{\mathrm{true}}/S\cdot S/(S-\eta)$ (which double-counts the $(S-\eta)^{-1}$ correction) is confirmed \emph{not} to equal the bias. Lean: the identity \texttt{claim3\_qtrue}, the three bias forms, nonnegativity exactly under $S>0$, $\eta\ge0$, $q_{\mathrm{true}}\ge0$ (theorem \texttt{claim3\_bias\_nonneg\_of\_qtrue}), the spurious-chain refutation, and exact rational bounds on the per-$\mu$ biases. Numeric: $S(1)=0.770$, $S(1.5)=0.691$, $S(2)=0.627$ at Gaza demographics. Direction note: the slack makes the paper's \emph{upper} bounds conservative in the claim-favourable direction, so no downstream conclusion depends on suppressing it.
\end{verification}

\section{Claim 4: manpower bound (paper Corollary 3.4)}

\begin{claim}\label{cl:manpower}
$q \le M/D$; whenever $\max\{0,q(\mu_{\mathrm{hi}})\}\le\min\{1,\,q(\mu_{\mathrm{lo}}),\,M/D\}$, the joint identified set is $\bigl[\max\{0,q(\mu_{\mathrm{hi}})\},\ \min\{1,\,q(\mu_{\mathrm{lo}}),\,M/D\}\bigr]$; otherwise it is empty and the inputs are jointly infeasible.
\end{claim}

\begin{proof}
Combatants killed cannot exceed combatants existing: $D_M\le M$. Divide by $D>0$. The joint set is the intersection of two intervals, hence the endpoint formulas. (If the intersection is empty, the inputs are jointly infeasible --- this is exactly the event detected by a positive contradiction radius.)
\end{proof}

\begin{verification}
\PASS\ (algebraic identity; Lean: \texttt{claim4\_manpower\_bound}). One caveat for manual review: $M$ here is the stock at $t_0$; if recruitment during the war is material, $M$ should be the stock-plus-inflow, which \emph{raises} the bound. In the Gaza application the bound is non-binding ($M/D=0.630$ vs.\ the demographic cap $q(1)=0.321$).
\end{verification}

\section{Claim 5: delta-method standard error (paper \S3)}

\begin{claim}\label{cl:delta}
For $\hat q = 1-\hat\omega(1-\hat f)/\hat w$ with independent inputs,
\[
\widehat{\mathrm{Var}}(\hat q)\approx
\frac{\hat\omega^2}{\hat w^2}\widehat{\mathrm{Var}}(\hat f)
+\frac{(1-\hat f)^2}{\hat w^2}\widehat{\mathrm{Var}}(\hat\omega)
+\frac{\hat\omega^2(1-\hat f)^2}{\hat w^4}\widehat{\mathrm{Var}}(\hat w).
\]
\end{claim}

\begin{proof}
First-order Taylor expansion of $g(\omega,f,w)=1-\omega(1-f)/w$:
\[
\partial_f g = \frac{\omega}{w},\qquad
\partial_\omega g = -\frac{1-f}{w},\qquad
\partial_w g = \frac{\omega(1-f)}{w^2}.
\]
$\mathrm{Var}(\hat q)\approx\sum(\partial g)^2\mathrm{Var}$, using independence (no covariance terms). Squaring the gradients gives the display.
\end{proof}

\begin{verification}
\PASS. All three partial derivatives and the assembled formula verified symbolically; the gradients are also kernel-checked as \texttt{HasDerivAt} statements in Lean (\texttt{claim5\_grad\_f/om/w}). Assumption to flag for manual review: independence of $(\hat\omega,\hat f,\hat w)$ is plausible here because they come from different institutions (deaths records, military-balance estimates, census), but any common-mode error (e.g.\ both $\omega$ and $w$ derived from the same census frame) would add covariance terms. The paper's conservative $\sigma_\omega=0.01$ (5$\times$ the binomial SE) partially absorbs this.
\end{verification}

\section{Claim 6: precision-weighted aggregation (paper Proposition 2.1)}

\begin{claim}
For independent unbiased sources with known variances $\sigma_i^2$ (with estimated $\hat\sigma_i^2$ independent of the point estimates the plug-in combination remains conditionally unbiased but not in general minimum-variance in finite samples; when the estimates are also consistent it is asymptotically efficient among linear unbiased combinations, under standard regularity), the estimator
\[
\hat\theta_{\mathrm{agg}}=\frac{\sum_i\hat\theta_i/\hat\sigma_i^2}{\sum_i 1/\hat\sigma_i^2}
\]
is minimum-variance among linear unbiased combinations with weights fixed given the variances. With per-source biases $b_i$, the aggregate bias is the precision-weighted mean of the $b_i$.
\end{claim}

\begin{proof}
\emph{(i)} Minimise $\sum c_i^2\hat\sigma_i^2$ subject to $\sum c_i=1$. The Lagrangian condition gives $2c_i\hat\sigma_i^2=\kappa$, i.e.\ $c_i\propto 1/\hat\sigma_i^2$; the constraint fixes the normalisation. The objective is strictly convex, so this is the unique minimum (Gauss--Markov in one dimension).

\emph{(ii)} $\E\hat\theta_{\mathrm{agg}}-\theta=\sum_i c_i b_i$ with $c_i=\hat\sigma_{\mathrm{agg}}^2/\hat\sigma_i^2$; direct substitution.
\end{proof}

\begin{verification}
\PASS. Optimality confirmed on 300 random configurations against random competitor weights; bias formula confirmed symbolically. In Lean, the precision weights are proved to sum to one and the minimum-variance inequality $\sum c_i^2\sigma_i^2\ge 1/\sum\sigma_i^{-2}$ is proved for every unbiased combination via Cauchy--Schwarz (\texttt{claim6\_optimality}).
\end{verification}

\section{Claim 7: feasibility and contradiction (paper Theorem 4.3)}

\begin{claim}
Fix $(\hat D,\hat N)$ and $[\mu_{\mathrm{lo}},\mu_{\mathrm{hi}}]$. For each independent input $A\in\{\omega,w,M\}$, revise that input alone, re-impose $a'=1-w'$ when $w$ moves and $f'=M'/\hat N$ when $M$ moves, and define
\[
\rho_A^{(i)}
:=\inf\left\{\frac{|A'-\hat A|}{\sigma_A}:
q^{(i)}\in\mathcal F\bigl(\hat x[A\mapsto A']\bigr)\right\},
\qquad
\rho^{(i)}:=\min_A\rho_A^{(i)}.
\]
Then (i) $\rho^{(i)}=0$ iff $q^{(i)}\in\mathcal F(\hat x)$, and (ii) every one-axis rescue on axis $A$ moves that input by at least $\rho_A^{(i)}\ge\rho^{(i)}$ standardised units.
\end{claim}

\begin{proof}
\emph{(i, $\Leftarrow$)} If $q^{(i)}\in\mathcal F(\hat x)$, the unmoved value $A'=\hat A$ is feasible on every axis, so every axis radius and hence $\rho^{(i)}$ is zero.

\emph{(i, $\Rightarrow$)} If $\rho^{(i)}=0$, some axis $A$ has $\rho_A^{(i)}=0$. Take feasible revisions $A'_n\to\hat A$. The re-imposed accounting identities converge with that axis. Each revision carries some $\mu_n$ in the compact exposure interval; pass to a subsequence with $\mu_n\to\mu^\ast$. Continuity of the demographic equation (with $w>0$) and of the manpower inequality gives both membership conditions of $\mathcal F(\hat x)$ in the limit. Thus $q^{(i)}\in\mathcal F(\hat x)$.

\emph{(ii)} Any feasible one-axis revision is an element of the set over which $\rho_A^{(i)}$ takes its infimum, so its standardised displacement is at least $\rho_A^{(i)}$, which is in turn at least $\min_B\rho_B^{(i)}=\rho^{(i)}$.
\end{proof}

\begin{verification}
\PASS. Kernel-checked ingredients (Lean): feasibility is monotone in $M$; $\omega_{\mathrm{needed}}$ is antitone in $\mu$; the radius lower-bounds every rationalisation's penalty and vanishes at a feasible face-value point. The compactness/continuity limit step is proved in the manuscript appendix. The code computes the $\omega$-axis radius as the distance from $\hat\omega$ to $[\omega_{\mathrm{needed}}(\mu_{\mathrm{hi}}),\omega_{\mathrm{needed}}(\mu_{\mathrm{lo}})]$, thereby granting the claim its most favourable admissible $\mu$.
\end{verification}

\section{Claim 8: deterministic source-deletion sensitivity (paper Proposition 5.1)}

\begin{claim}
Let $\Theta_\varnothing=[L_0,U_0]$ be the compact face-value identified set. Deleting sources in $T$ gives a compact set $\Theta_T\supseteq\Theta_\varnothing$ of diameter $R_T$; every nonempty identified set $\widetilde\Theta_T$ obtained by replacing those sources while retaining all others satisfies $\widetilde\Theta_T\subseteq\Theta_T$. Then (i) $\widetilde\Theta_T\subseteq[L_0-R_T,U_0+R_T]$. (ii) If $d_T^-$ and $d_T^+$ are the worst downward and upward endpoint displacements over replacements, their deterministic multilinear extensions
\[
\Delta_s(\varepsilon)=\sum_{T\subseteq\mathcal I}
\prod_{i\in T}\varepsilon_i\prod_{j\notin T}(1-\varepsilon_j)d_T^s
\quad(s\in\{-,+\})
\]
satisfy $\Delta_s(\varepsilon)=\sum_i\varepsilon_i d_{\{i\}}^s+O(\|\varepsilon\|_\infty^2)\le\sum_i\varepsilon_iR_i+C\|\varepsilon\|_\infty^2$ for a finite constant $C$ determined by the deletion sets. Hence $[L_0-\sum_i\varepsilon_iR_i,U_0+\sum_i\varepsilon_iR_i]$ is a conservative deterministic first-order sensitivity band, while part (i) governs a specified finite deletion pattern.
\end{claim}

\begin{proof}
\emph{(i)} Since $L_0,U_0\in\Theta_T$ and every $x\in\widetilde\Theta_T$ also lies in $\Theta_T$, both $|x-L_0|$ and $|x-U_0|$ are at most $\operatorname{diam}(\Theta_T)=R_T$. This gives the stated containment and, after taking endpoint and replacement suprema, $0\le d_T^s\le R_T$.

\emph{(ii)} Separate the singleton terms in the defining polynomial. For $T=\{i\}$, the coefficient is $\varepsilon_i\prod_{j\ne i}(1-\varepsilon_j)=\varepsilon_i+O(\|\varepsilon\|_\infty^2)$. Every term with $|T|\ge2$ is $O(\|\varepsilon\|_\infty^2)$; there are finitely many and $d_T^s\le R_T<\infty$. With $n=|\mathcal I|$, $\delta=\|\varepsilon\|_\infty$, and $R_{\max}=\max_{T\ne\varnothing}R_T$, the remainder is bounded in absolute value by $[n(n-1)+2^n]R_{\max}\delta^2$. Thus $\Delta_s=\sum_i\varepsilon_i d_{\{i\}}^s+O(\delta^2)$, and $d_{\{i\}}^s\le R_i$ gives the conservative linear bound. Joint-removal diameters need not be bounded by sums of singleton diameters, so the finite-pattern claim remains part (i), not the first-order band.
\end{proof}

\begin{verification}
\PASS. The containment is exercised on 1{,}000 randomly generated nested intervals and replacement subsets. The directional derivative of the multilinear deletion functional at the origin is checked in 100 random configurations, together with its singleton-diameter bound. Downstream use in Section 6 labels the quoted $32.1\%\to39.4\%$ inflation as deterministic first-order source-deletion sensitivity.
\end{verification}
